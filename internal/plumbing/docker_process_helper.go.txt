// This host-owned program transports bytes and process state, never evidence.
// It is compiled by the prepared image's Go toolchain using only its stdlib.
package main

import (
	"context"
	"encoding/binary"
	"errors"
	"fmt"
	"io"
	"os"
	"os/exec"
	"runtime"
	"strconv"
	"sync"
	"syscall"
	"time"
)

const protocol = "invariant-process-v1\n"

type frames struct {
	mu  sync.Mutex
	err error
}

func (f *frames) write(kind byte, data []byte) error {
	f.mu.Lock()
	defer f.mu.Unlock()
	if f.err != nil {
		return f.err
	}
	header := [5]byte{kind}
	binary.BigEndian.PutUint32(header[1:], uint32(len(data)))
	_, f.err = os.Stdout.Write(header[:])
	if f.err == nil {
		_, f.err = os.Stdout.Write(data)
	}
	return f.err
}

func main() {
	if _, err := io.WriteString(os.Stdout, protocol); err != nil {
		os.Exit(125)
	}
	f := new(frames)
	if err := run(f); err != nil {
		_ = f.write('F', []byte(err.Error()))
		os.Exit(125)
	}
	if err := f.write('D', nil); err != nil {
		os.Exit(125)
	}
}

func run(f *frames) error {
	if len(os.Args) < 3 {
		return errors.New("process helper needs an endpoint variable and executable")
	}
	if err := adopt(); err != nil {
		return err
	}
	cmd := exec.Command(os.Args[2], os.Args[3:]...)
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.Env = os.Environ()
	var reads, writes []*os.File
	defer func() {
		for _, file := range append(reads, writes...) {
			_ = file.Close()
		}
	}()
	count := 2
	if os.Args[1] != "" {
		count++
	}
	for i := 0; i < count; i++ {
		r, w, err := os.Pipe()
		if err != nil {
			return err
		}
		reads, writes = append(reads, r), append(writes, w)
	}
	cmd.Stdout, cmd.Stderr = writes[0], writes[1]
	if count == 3 {
		cmd.ExtraFiles = []*os.File{writes[2]}
		cmd.Env = append(cmd.Env, os.Args[1]+"=/dev/fd/3")
	}
	if err := cmd.Start(); err != nil {
		return err
	}
	for _, file := range writes {
		_ = file.Close()
	}
	startErr := f.write('S', nil)
	stop := make(chan struct{}, 1)
	go func() {
		// The host alone owns stdin. EOF requests cancellation, and a lost
		// Docker connection cannot leave a workload waiting for more input.
		var b [1]byte
		_, _ = os.Stdin.Read(b[:])
		stop <- struct{}{}
	}()
	drains := make(chan error, count)
	for i, file := range reads {
		go func() {
			buf := make([]byte, 32<<10)
			for {
				n, err := file.Read(buf)
				if n > 0 {
					if writeErr := f.write("OEV"[i], buf[:n]); writeErr != nil {
						drains <- writeErr
						select {
						case stop <- struct{}{}:
						default:
						}
						return
					}
				}
				if err != nil {
					if errors.Is(err, io.EOF) {
						err = nil
					}
					drains <- err
					return
				}
			}
		}()
	}
	wait := make(chan error, 1)
	go func() { wait <- cmd.Wait() }()
	var exitErr error
	err := startErr
	exited := false
	if err == nil {
		select {
		case exitErr = <-wait:
			exited = true
		case <-stop:
			err = errors.New("process transport interrupted")
		}
	}
	// Normal exit also owns silent descendants, even if every pipe reached EOF.
	if killErr := syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL); killErr != nil && !errors.Is(killErr, syscall.ESRCH) {
		err = errors.Join(err, killErr)
	}
	cleanup, done := context.WithTimeout(context.Background(), 2*time.Second)
	defer done()
	if !exited {
		select {
		case exitErr = <-wait:
			exited = true
		case <-cleanup.Done():
			err = errors.Join(err, errors.New("workload did not exit"), cleanup.Err())
		}
	}
	if exited {
		var ordinary *exec.ExitError
		if exitErr != nil && !errors.As(exitErr, &ordinary) {
			err = errors.Join(err, exitErr)
		}
		err = errors.Join(err, f.write('X', []byte(strconv.Itoa(cmd.ProcessState.ExitCode()))))
	}
	err = errors.Join(err, reap(cleanup, cmd.Process.Pid))
	for remaining := count; remaining > 0; remaining-- {
		select {
		case drainErr := <-drains:
			err = errors.Join(err, drainErr)
		case <-cleanup.Done():
			err = errors.Join(err, errors.New("workload pipes did not drain"), cleanup.Err())
			for _, file := range reads {
				_ = file.Close()
			}
			for ; remaining > 0; remaining-- {
				err = errors.Join(err, <-drains)
			}
		}
	}
	return err
}

// Linux adopts only this helper's descendants; waits stay in its workload's
// group. Darwin is supported for daemon-free boundary tests and reaps via init.
func adopt() error {
	if runtime.GOOS == "darwin" {
		return nil
	}
	if runtime.GOOS != "linux" {
		return errors.New("unsupported process platform")
	}
	var number uintptr
	switch runtime.GOARCH {
	case "amd64":
		number = 157
	case "arm64", "riscv64":
		number = 167
	default:
		return errors.New("unsupported process reaper architecture")
	}
	_, _, errno := syscall.Syscall6(number, 36, 1, 0, 0, 0, 0)
	if errno != 0 {
		return errno
	}
	return nil
}

func reap(ctx context.Context, group int) error {
	for {
		var status syscall.WaitStatus
		for {
			pid, err := syscall.Wait4(-group, &status, syscall.WNOHANG, nil)
			if err != nil && !errors.Is(err, syscall.ECHILD) && !errors.Is(err, syscall.EINTR) {
				return err
			}
			if pid <= 0 {
				break
			}
		}
		if err := syscall.Kill(-group, 0); errors.Is(err, syscall.ESRCH) {
			return nil
		} else if err != nil {
			return err
		}
		if err := ctx.Err(); err != nil {
			return fmt.Errorf("process group was not reaped: %w", err)
		}
		_ = syscall.Kill(-group, syscall.SIGKILL)
		time.Sleep(time.Millisecond)
	}
}
