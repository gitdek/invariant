// This standard-library program is compiled by the prepared image. It knows
// process channels and status only; framework protocols remain opaque bytes.
package main

import (
	"encoding/binary"
	"errors"
	"fmt"
	"io"
	"os"
	"os/exec"
	"runtime"
	"strconv"
	"sync"
	"syscall"
	"time"
)

type connection struct {
	mu     sync.Mutex
	err    error
	broken chan struct{}
}

func (c *connection) frame(kind byte, data []byte) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.err != nil {
		return c.err
	}
	var header [5]byte
	header[0] = kind
	binary.BigEndian.PutUint32(header[1:], uint32(len(data)))
	_, err := os.Stdout.Write(header[:])
	if err == nil {
		_, err = os.Stdout.Write(data)
	}
	if err != nil {
		c.err = err
		close(c.broken)
	}
	return err
}

func main() {
	c := &connection{broken: make(chan struct{})}
	if _, err := io.WriteString(os.Stdout, "invariant-process-v1\n"); err != nil {
		os.Exit(1)
	}
	if err := run(c); err != nil {
		text := []byte(err.Error())
		if len(text) > 64<<10 {
			text = text[:64<<10]
		}
		_ = c.frame('f', text)
		os.Exit(1)
	}
}

func run(c *connection) error {
	if len(os.Args) < 3 {
		return errors.New("process helper needs an endpoint variable and arguments")
	}
	if err := adopt(); err != nil {
		return err
	}
	var reads, writes []*os.File
	defer func() {
		for _, file := range append(reads, writes...) {
			_ = file.Close()
		}
	}()
	for range 3 {
		r, w, err := os.Pipe()
		if err != nil {
			return err
		}
		reads, writes = append(reads, r), append(writes, w)
	}
	cmd := exec.Command(os.Args[2], os.Args[3:]...)
	cmd.Env = os.Environ()
	cmd.Stdout, cmd.Stderr = writes[0], writes[1]
	if os.Args[1] != "" {
		cmd.ExtraFiles = []*os.File{writes[2]}
		cmd.Env = append(cmd.Env, os.Args[1]+"=/dev/fd/3")
	}
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	if err := cmd.Start(); err != nil {
		return err
	}
	for _, w := range writes {
		_ = w.Close()
	}
	startErr := c.frame('s', nil)
	drained := make(chan error, len(reads))
	for i, file := range reads {
		go func() {
			defer file.Close()
			buf := make([]byte, 32<<10)
			for {
				n, err := file.Read(buf)
				if n > 0 {
					if writeErr := c.frame("oev"[i], buf[:n]); writeErr != nil {
						drained <- writeErr
						return
					}
				}
				if err != nil {
					if errors.Is(err, io.EOF) {
						err = nil
					}
					drained <- err
					return
				}
			}
		}()
	}
	wait := make(chan error, 1)
	go func() { wait <- cmd.Wait() }()
	stop := make(chan struct{})
	go func() {
		_, _ = io.Copy(io.Discard, os.Stdin)
		close(stop)
	}()
	defer os.Stdin.Close()
	var exitErr error
	err := startErr
	finished := false
	select {
	case exitErr = <-wait:
		finished = true
	case <-c.broken:
		err = errors.Join(err, errors.New("process connection broke"))
	case <-stop:
		err = errors.Join(err, errors.New("process interrupted by host"))
	}
	// Pipe EOF alone does not establish ownership cleanup: a silent descendant
	// may still be alive. Terminate and reap the workload's entire group first.
	// Darwin answers EPERM while a group's last members are reaped, and reaping
	// still waits for the group to disappear (#516).
	if killErr := syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL); killErr != nil && !errors.Is(killErr, syscall.ESRCH) && !errors.Is(killErr, syscall.EPERM) {
		err = errors.Join(err, killErr)
	}
	end := time.Now().Add(2 * time.Second)
	if !finished {
		exitErr = <-wait
	}
	var ordinary *exec.ExitError
	if exitErr != nil && !errors.As(exitErr, &ordinary) {
		err = errors.Join(err, exitErr)
	}
	if cmd.ProcessState == nil {
		err = errors.Join(err, errors.New("process has no exit state"))
	} else {
		err = errors.Join(err, c.frame('x', []byte(strconv.Itoa(cmd.ProcessState.ExitCode()))))
	}
	err = errors.Join(err, reap(cmd.Process.Pid, end))
	timer := time.NewTimer(max(0, time.Until(end)))
	defer timer.Stop()
	for remaining := len(reads); remaining > 0; remaining-- {
		select {
		case drainErr := <-drained:
			err = errors.Join(err, drainErr)
		case <-timer.C:
			err = errors.Join(err, errors.New("process channels did not drain"))
			for _, file := range reads {
				_ = file.Close()
			}
			for ; remaining > 0; remaining-- {
				err = errors.Join(err, <-drained)
			}
		}
	}
	if err != nil {
		return err
	}
	return c.frame('d', nil)
}

func adopt() error {
	if runtime.GOOS == "darwin" {
		return nil
	}
	if runtime.GOOS != "linux" {
		return errors.New("process helper requires Linux or Darwin")
	}
	var number uintptr
	switch runtime.GOARCH {
	case "amd64":
		number = 157
	case "arm64", "riscv64":
		number = 167
	default:
		return errors.New("unsupported process reaper architecture")
	}
	_, _, errno := syscall.Syscall6(number, 36, 1, 0, 0, 0, 0)
	if errno != 0 {
		return fmt.Errorf("adopt process descendants: %w", errno)
	}
	return nil
}

func reap(group int, end time.Time) error {
	for {
		var status syscall.WaitStatus
		for {
			pid, err := syscall.Wait4(-group, &status, syscall.WNOHANG, nil)
			if err != nil && !errors.Is(err, syscall.ECHILD) && !errors.Is(err, syscall.EINTR) {
				return err
			}
			if pid <= 0 {
				break
			}
		}
		if err := syscall.Kill(-group, 0); errors.Is(err, syscall.ESRCH) {
			return nil
		} else if err != nil && !errors.Is(err, syscall.EPERM) {
			return err
		}
		if !time.Now().Before(end) {
			return errors.New("process descendants did not exit")
		}
		_ = syscall.Kill(-group, syscall.SIGKILL)
		time.Sleep(time.Millisecond)
	}
}
