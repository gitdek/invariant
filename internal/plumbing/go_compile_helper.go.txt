type binding struct {
	InvocationID, Part, Framework, File, Selector, Package, SourceDigest, ExecutionDigest string
}

type request struct {
	Binding               binding
	File, Package, Source string
	Flags, Argv           []string
}

type report struct {
	Binding         binding
	ExitCode        int
	Reason, Problem string
}

type packageInfo struct {
	Dir, ImportPath, Name, Export                                                 string
	GoFiles, CgoFiles, TestGoFiles, XTestGoFiles, TestImports, XTestImports, Deps []string
	Incomplete                                                                    bool
	Error                                                                         *struct{ Err string }
	Module                                                                        *struct {
		Path, GoVersion string
		Main            bool
	}
}

func main() {
	endpoint := os.Getenv("INVARIANT_GO_COMPILE_EVENTS")
	if err := os.Unsetenv("INVARIANT_GO_COMPILE_EVENTS"); err != nil {
		panic(err)
	}
	events, err := os.OpenFile(endpoint, os.O_WRONLY, 0)
	if err != nil {
		panic(err)
	}
	defer events.Close()
	if value, ok := strings.CutPrefix(endpoint, "/dev/fd/"); ok {
		fd, err := strconv.Atoi(value)
		if err != nil || fd < 3 {
			panic("invalid compiler event descriptor")
		}
		if err := os.NewFile(uintptr(fd), endpoint).Close(); err != nil {
			panic(err)
		}
	}
	data, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	var in request
	if err := json.Unmarshal(data, &in); err != nil {
		panic(err)
	}
	out := report{Binding: in.Binding, ExitCode: -1}
	if err := analyze(in, &out); err != nil {
		out.Problem = err.Error()
		out.Reason = ""
	}
	if err := json.NewEncoder(events).Encode(out); err != nil {
		panic(err)
	}
}

func goCommand(args ...string) *exec.Cmd {
	return exec.Command(filepath.Join(runtime.GOROOT(), "bin", "go"), args...)
}

func list(flags []string, export bool, packages ...string) ([]packageInfo, error) {
	args := []string{"list", "-json"}
	if export {
		args = append(args, "-deps", "-export")
	}
	args = append(args, flags...)
	args = append(args, packages...)
	cmd := goCommand(args...)
	cmd.Stderr = os.Stderr
	data, err := cmd.Output()
	if err != nil {
		return nil, fmt.Errorf("ordinary package loading or compilation failed: %w", err)
	}
	var result []packageInfo
	decoder := json.NewDecoder(bytes.NewReader(data))
	for {
		var pkg packageInfo
		if err := decoder.Decode(&pkg); err != nil {
			if err == io.EOF {
				break
			}
			return nil, err
		}
		if pkg.Incomplete || pkg.Error != nil || pkg.ImportPath == "" {
			return nil, errors.New("incomplete ordinary package metadata")
		}
		result = append(result, pkg)
	}
	return result, nil
}

// Read only regular sources beneath the prepared part, as the host reader does.
func read(root *os.Root, name string) ([]byte, error) {
	if !filepath.IsLocal(name) || filepath.Clean(name) != name {
		return nil, errors.New("invalid source path")
	}
	parts := strings.Split(name, string(filepath.Separator))
	for i := 1; i < len(parts); i++ {
		info, err := root.Lstat(filepath.Join(parts[:i]...))
		if err != nil {
			return nil, err
		}
		if !info.IsDir() {
			return nil, errors.New("unsafe source ancestor")
		}
	}
	before, err := root.Lstat(name)
	if err != nil {
		return nil, err
	}
	if !before.Mode().IsRegular() {
		return nil, errors.New("source is not regular")
	}
	f, err := root.OpenFile(name, os.O_RDONLY|syscall.O_NOFOLLOW|syscall.O_NONBLOCK, 0)
	if err != nil {
		return nil, err
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil {
		return nil, err
	}
	if !os.SameFile(before, info) || !info.Mode().IsRegular() || info.Sys().(*syscall.Stat_t).Nlink != 1 {
		return nil, errors.New("source changed or is linked")
	}
	return io.ReadAll(f)
}

func analyze(in request, out *report) error {
	if strings.Contains(in.Binding.Selector, "/") {
		return errors.New("compiler attribution requires a top-level selector; child identity needs runtime evidence")
	}
	root, err := os.OpenRoot(".")
	if err != nil {
		return err
	}
	defer root.Close()
	pinned, err := read(root, in.File)
	if err != nil {
		return err
	}
	if string(pinned) != in.Source || fmt.Sprintf("sha256:%x", sha256.Sum256(pinned)) != in.Binding.SourceDigest {
		return errors.New("pinned compiler source changed")
	}
	loaded, err := list(in.Flags, false, in.Package)
	if err != nil {
		return err
	}
	if len(loaded) != 1 {
		return errors.New("compiler analysis needs one package")
	}
	pkg := loaded[0]
	dir, err := filepath.Abs(filepath.Dir(in.File))
	if err != nil {
		return err
	}
	if pkg.Dir != dir || pkg.ImportPath != in.Binding.Package || pkg.Module == nil || !pkg.Module.Main || len(pkg.CgoFiles) != 0 || len(pkg.GoFiles) == 0 {
		return errors.New("compiler analysis needs ordinary production Go files in the discovered package")
	}
	positions := token.NewFileSet()
	var production, internal, external []*ast.File
	sources := map[string][]byte{}
	productionFiles := map[string]bool{}
	for group, names := range [][]string{pkg.GoFiles, pkg.TestGoFiles, pkg.XTestGoFiles} {
		for _, base := range names {
			if filepath.Base(base) != base || !strings.HasSuffix(base, ".go") {
				return errors.New("invalid included Go file")
			}
			name := filepath.Join(filepath.Dir(in.File), base)
			if _, exists := sources[name]; exists {
				return errors.New("duplicate included Go file")
			}
			data, err := read(root, name)
			if err != nil {
				return err
			}
			sources[name] = data
			parsed, err := parser.ParseFile(positions, name, data, parser.ParseComments|parser.AllErrors)
			if err != nil {
				return fmt.Errorf("invalid included source: %w", err)
			}
			want := pkg.Name
			if group == 2 {
				want += "_test"
			}
			if parsed.Name.Name != want {
				return errors.New("included source has a different package")
			}
			if group != 0 {
				for _, comments := range parsed.Comments {
					for _, comment := range comments.List {
						if strings.HasPrefix(comment.Text, "//go:") && !strings.HasPrefix(comment.Text, "//go:build ") {
							return errors.New("test compiler directives are outside conservative attribution")
						}
					}
				}
				for _, decl := range parsed.Decls {
					if fn, ok := decl.(*ast.FuncDecl); ok && fn.Body == nil {
						return errors.New("included test function has no body")
					}
				}
			}
			switch group {
			case 0:
				production = append(production, parsed)
				productionFiles[name] = true
			case 1:
				internal = append(internal, parsed)
			case 2:
				external = append(external, parsed)
			}
		}
	}
	if string(sources[in.File]) != in.Source {
		return errors.New("pinned file is not included unchanged")
	}
	// Test-package setup spans internal and external tests. Let the prepared
	// toolchain validate it before a compiler failure can supply authority.
	if _, err := list(append(slices.Clone(in.Flags), "-test"), false, in.Package); err != nil {
		return fmt.Errorf("test package setup failed: %w", err)
	}

	// Compile production and all test dependencies normally. Import their real
	// export data; never stub an API, synthesize a package, or filter production.
	imports := append([]string{in.Package}, pkg.TestImports...)
	imports = append(imports, pkg.XTestImports...)
	dependencies, err := list(in.Flags, true, imports...)
	if err != nil {
		return err
	}
	exports := map[string]string{}
	for _, dep := range dependencies {
		if dep.Export != "" {
			exports[dep.ImportPath] = dep.Export
		}
		if slices.Contains(pkg.TestImports, dep.ImportPath) && (dep.ImportPath == pkg.ImportPath || slices.Contains(dep.Deps, pkg.ImportPath)) {
			return errors.New("ordinary tests introduce an import cycle")
		}
	}
	if exports[pkg.ImportPath] == "" {
		return errors.New("production package has no compiled export data")
	}
	lookup := func(name string) (io.ReadCloser, error) {
		file := exports[name]
		if file == "" {
			return nil, fmt.Errorf("no prepared export for %s", name)
		}
		return os.Open(file)
	}
	compiled := goCommand(in.Argv...)
	compiled.Stdout, compiled.Stderr = os.Stdout, os.Stderr
	err = compiled.Run()
	var exit *exec.ExitError
	if !errors.As(err, &exit) || compiled.ProcessState == nil || exit.ExitCode() <= 0 || exit.ExitCode() >= 125 {
		return fmt.Errorf("no completed failing compiler process: %v", err)
	}
	out.ExitCode = exit.ExitCode()
	var reasons []string
	for group, tests := range [][]*ast.File{internal, external} {
		if len(tests) == 0 {
			continue
		}
		files := tests
		name := pkg.ImportPath + "_test"
		if group == 0 {
			files = append(slices.Clone(production), tests...)
			name = pkg.ImportPath
		}
		info := &types.Info{Types: map[ast.Expr]types.TypeAndValue{}, Defs: map[*ast.Ident]types.Object{}, Uses: map[*ast.Ident]types.Object{}, Scopes: map[ast.Node]*types.Scope{}}
		var failures []types.Error
		config := types.Config{Importer: importer.ForCompiler(positions, "gc", lookup), Sizes: types.SizesFor("gc", runtime.GOARCH), Error: func(err error) {
			if typed, ok := err.(types.Error); ok {
				failures = append(failures, typed)
			} else {
				failures = append(failures, types.Error{})
			}
		}}
		if pkg.Module.GoVersion != "" {
			config.GoVersion = "go" + pkg.Module.GoVersion
		}
		checked, checkErr := config.Check(name, positions, files, info)
		if checkErr != nil && len(failures) == 0 {
			return errors.New("type analysis failed without source positions")
		}
		uses := map[token.Pos]string{}
		selected := map[token.Pos]bool{}
		for _, file := range tests {
			for _, decl := range file.Decls {
				fn, ok := decl.(*ast.FuncDecl)
				if !ok || fn.Recv != nil {
					continue
				}
				kind := testKind(fn.Name.Name)
				if kind == "" {
					continue
				}
				if !testingFunction(fn, info, kind) {
					return fmt.Errorf("invalid included testing declaration %s", fn.Name.Name)
				}
				if kind != "T" || positions.PositionFor(file.Pos(), false).Filename != in.File {
					continue
				}
				ast.Inspect(fn.Body, func(node ast.Node) bool {
					if _, nested := node.(*ast.FuncLit); nested {
						return false
					}
					var expr ast.Expr
					var literalType bool
					switch n := node.(type) {
					case *ast.CallExpr:
						for _, arg := range n.Args {
							if testingValue(info.TypeOf(arg)) {
								return true // A missing testing helper is not a production API.
							}
						}
						expr = n.Fun
					case *ast.CompositeLit:
						expr = n.Type
						literalType = true
					}
					id := proposedAPI(expr, literalType, info, checked, pkg.ImportPath, group == 0, positions, productionFiles)
					if id != nil {
						at := positions.PositionFor(id.Pos(), false)
						uses[id.Pos()] = fmt.Sprintf("%s:%d:%d: %s directly uses proposed production API %s", in.File, at.Line, at.Column, fn.Name.Name, id.Name)
						selected[id.Pos()] = fn.Name.Name == in.Binding.Selector
					}
					return true
				})
			}
		}
		for _, failure := range failures {
			reason := uses[failure.Pos]
			if reason == "" {
				return fmt.Errorf("type error outside attributable production API uses: %s", failure.Error())
			}
			if selected[failure.Pos] {
				reasons = append(reasons, reason)
			}
		}
	}
	if len(reasons) == 0 {
		return fmt.Errorf("%s has no direct attributable production API error in %s", in.Binding.Selector, in.File)
	}
	// The evidence describes the same sources and inclusion policy the compiler
	// and analyzer consumed, even when a prior command changed private storage.
	again, err := list(in.Flags, false, in.Package)
	if err != nil {
		return err
	}
	if !reflect.DeepEqual(loaded, again) {
		return errors.New("package inclusion changed during compiler attribution")
	}
	for name, before := range sources {
		after, err := read(root, name)
		if err != nil {
			return err
		}
		if !bytes.Equal(before, after) {
			return fmt.Errorf("source changed during compiler attribution: %s", name)
		}
	}
	out.Reason = "Go compilation failed: " + strings.Join(reasons, "; ")
	return nil
}

func testKind(name string) string {
	if name == "TestMain" {
		return "M"
	}
	for _, prefix := range []struct{ name, kind string }{{"Test", "T"}, {"Benchmark", "B"}, {"Fuzz", "F"}, {"Example", "example"}} {
		if rest, ok := strings.CutPrefix(name, prefix.name); ok {
			r, _ := utf8.DecodeRuneInString(rest)
			if rest == "" || !unicode.IsLower(r) {
				return prefix.kind
			}
		}
	}
	return ""
}

func testingFunction(fn *ast.FuncDecl, info *types.Info, kind string) bool {
	object, ok := info.Defs[fn.Name].(*types.Func)
	if !ok {
		return false
	}
	sig, ok := object.Type().(*types.Signature)
	if !ok || sig.TypeParams().Len() != 0 || sig.Results().Len() != 0 || sig.Variadic() {
		return false
	}
	if kind == "example" {
		return sig.Params().Len() == 0
	}
	if sig.Params().Len() != 1 {
		return false
	}
	ptr, ok := sig.Params().At(0).Type().(*types.Pointer)
	if !ok {
		return false
	}
	named, ok := ptr.Elem().(*types.Named)
	return ok && named.Obj().Pkg() != nil && named.Obj().Pkg().Path() == "testing" && named.Obj().Name() == kind
}

func proposedAPI(expr ast.Expr, literalType bool, info *types.Info, pkg *types.Package, production string, internal bool, positions *token.FileSet, productionFiles map[string]bool) *ast.Ident {
	switch expr := expr.(type) {
	case *ast.Ident:
		if !internal || !ast.IsExported(expr.Name) || info.Uses[expr] != nil || info.Defs[expr] != nil {
			return nil
		}
		for _, scope := range info.Scopes {
			if scope.Contains(expr.Pos()) {
				if _, object := scope.LookupParent(expr.Name, token.NoPos); object != nil {
					return nil
				}
			}
		}
		return expr
	case *ast.SelectorExpr:
		if !ast.IsExported(expr.Sel.Name) || info.Uses[expr.Sel] != nil {
			return nil
		}
		if id, ok := expr.X.(*ast.Ident); ok {
			if object, ok := info.Uses[id].(*types.PkgName); ok {
				if object.Imported().Path() == production && object.Imported().Scope().Lookup(expr.Sel.Name) == nil {
					return expr.Sel
				}
				return nil
			}
		}
		if literalType {
			return nil // Only a package can qualify a composite-literal type.
		}
		typ := info.TypeOf(expr.X)
		if typ == nil {
			return nil
		}
		object, index, _ := types.LookupFieldOrMethod(typ, true, pkg, expr.Sel.Name)
		if object != nil || len(index) != 0 {
			return nil
		}
		typ = types.Unalias(typ)
		if ptr, ok := typ.(*types.Pointer); ok {
			typ = types.Unalias(ptr.Elem())
			if _, ok := typ.Underlying().(*types.Interface); ok {
				return nil // A pointer to an interface cannot select its methods.
			}
		}
		named, ok := typ.(*types.Named)
		if !ok || named.Obj().Pkg() == nil || named.Obj().Pkg().Path() != production {
			return nil
		}
		if internal && !productionFiles[positions.PositionFor(named.Obj().Pos(), false).Filename] {
			return nil
		}
		return expr.Sel
	}
	return nil
}

func testingValue(typ types.Type) bool {
	if typ == nil {
		return false
	}
	typ = types.Unalias(typ)
	if ptr, ok := typ.(*types.Pointer); ok {
		typ = types.Unalias(ptr.Elem())
	}
	named, ok := typ.(*types.Named)
	return ok && named.Obj().Pkg() != nil && named.Obj().Pkg().Path() == "testing"
}
